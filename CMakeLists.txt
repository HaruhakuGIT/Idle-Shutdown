cmake_minimum_required(VERSION 3.24)
project(ShutdownWarning VERSION 1.0.8 LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)

if(WIN32)
    enable_language(RC)
    configure_file(app/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    add_executable(ShutdownWarning WIN32 app/windows/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
    target_compile_features(ShutdownWarning PRIVATE cxx_std_20)
    target_include_directories(ShutdownWarning PRIVATE app/common app/windows)
    target_compile_definitions(ShutdownWarning PRIVATE UNICODE _UNICODE)
    target_link_libraries(ShutdownWarning PRIVATE d2d1 dwrite winmm user32 gdi32 shell32 advapi32 comctl32 dwmapi)
    if(MSVC)
        set_property(TARGET ShutdownWarning PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(ShutdownWarning PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
    endif()
    install(TARGETS ShutdownWarning RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(app/linux)
else()
    message(FATAL_ERROR "Idle Shutdown supports Windows and Linux.")
endif()

if(BUILD_TESTING)
    add_executable(shutdown-warning-sound-test tests/sound_test.cpp)
    target_compile_features(shutdown-warning-sound-test PRIVATE cxx_std_20)
    target_include_directories(shutdown-warning-sound-test PRIVATE app/common)
    add_test(NAME synthesized-audio COMMAND shutdown-warning-sound-test)

    add_executable(shutdown-warning-motion-test tests/mouse_motion_test.cpp)
    target_compile_features(shutdown-warning-motion-test PRIVATE cxx_std_20)
    target_include_directories(shutdown-warning-motion-test PRIVATE app/common)
    add_test(NAME mouse-motion COMMAND shutdown-warning-motion-test)
endif()
